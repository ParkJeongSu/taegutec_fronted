└─ src
   ├─ api/              # API 호출 함수 (Business Logic과 분리)
   ├─ assets/           # 이미지, 전역 CSS, 폰트
   ├─ components/       # 재사용 가능한 부품들
   │  ├─ common/        # 버튼, 인풋, 모달 등 (비즈니스 로직 없음)
   │  └─ widgets/       # 특정 기능 단위 (예: ChartWidget, UserProfileCard)
   ├─ composables/      # 상태나 공통 로직 (Setup API에서 활용)
   ├─ constants/        # 공통 상수 (에러 메시지, 코드 값 등)
   ├─ layouts/          # 전체 페이지 틀 (MainLayout.vue, AuthLayout.vue)
   │  └─ parts/         # 레이아웃 전용 부품 (Header.vue, Sidebar.vue)
   ├─ locales/          # 다국어 정의
   ├─ plugins/          # 다국어 설정 및 vuetify 설정 plugins
   ├─ router/           # 라우터 설정
   ├─ stores/           # 상태 관리 (Pinia 등)
   ├─ utils/            # 순수 자바스크립트 유틸 함수 (날짜 계산, 포맷팅)
   ├─ views/            # 라우트와 연결된 페이지 단위
   │  └─ Production/    # views 하위 카테고리별 화면 모음 추가 변경 가능
   │    └─ components/  # Production 하위의 자식 화면 모음
   │  └─ History/       # views 하위 카테고리별 화면 모음 추가 변경 가능
   │    └─ components/  # History 하위의 자식 화면 모음
   │  └─ Definition/    # views 하위 카테고리별 화면 모음 추가 변경 가능
   │    └─ components/  # Definition 하위의 자식 화면 모음
   ├─ App.vue           # App.vue
   └─ main.js           # main.js
